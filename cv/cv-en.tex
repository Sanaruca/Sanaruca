% =========================================================
% 3. CUSTOM COMMANDS
% =========================================================

% Section titles in the left column
\newcommand{\CVSection}[1]{
    \vspace{0.1cm}
    \noindent{\large\bfseries\ttfamily\color{CVPrimary} \raggedright \MakeUppercase{#1}\par}
    \vspace{0.5cm}
}

% Section titles in "Normal" (full width) mode
\newcommand{\CVSectionNormal}[1]{
    \vspace{1cm}
    \noindent{\large\bfseries\ttfamily\color{CVPrimary} \MakeUppercase{#1}}\par
    \vspace{0.5cm}
}

% Simple text block (profile) with side line
\newcommand{\CVTextBlock}[1]{
    \noindent
    \textcolor{CVLine}{\vrule width 1pt} \hspace{1cm}
    \begin{minipage}[t]{0.95\linewidth}
        \raggedright {\DMSansLight #1}
    \end{minipage}
    \vspace{0.3cm}
}

% Education with side line
\newcommand{\CVEdu}[3]{
    \noindent
    \textcolor{CVLine}{\vrule width 1pt} \hspace{1cm}
    \begin{minipage}[t]{0.95\linewidth}
        \textbf{#1} \hfill \textbf{\color{CVPrimary}#3} \\
        \textit{#2}
    \end{minipage}
    \vspace{0.3cm}
}

% References
\newcommand{\CVRef}[4]{
    \noindent
    \textcolor{CVLine}{\vrule width 1pt} \hspace{1cm}
    \begin{minipage}[t]{0.42\textwidth}
        {\PlexMedium #1} \\
        {\PlexSans #2} \\
        {\PlexLight #3} \\
        {\DMSansMedium tel.}{\DMSansLight\ #4}
    \end{minipage}
}

% Experience metadata block in the left column
% Exact order: role / company / location / period (stacked)
\newcommand{\CVExpMeta}[4]{
    {\noindent
    {\PlexMedium\color{CVPrimary} #1} \\
    {\PlexSans #2} \\
    {\PlexLight\itshape #3} \\
    {\PlexExtraLight #4}}
    \vspace{0.85cm}
}

% Description body in the right column of Experience
\newcommand{\CVExpBody}[1]{
    \noindent
    \textcolor{CVLine}{\vrule width 1pt} \hspace{1cm}
    \begin{minipage}[t]{0.92\linewidth}
        \raggedright {\DMSansLight #1}
    \end{minipage}
    \vspace{0.8cm}
}

% Contact entry: fixed-width abbreviation (Medium) + value (Light)
\newcommand{\CVContact}[2]{
    {\DMSansMedium\makebox[1.1cm][l]{#1}} {\DMSansLight #2} \\[0.04cm]
}

% Education item: title / issuing entity / location / date (stacked)
\newcommand{\CVEdItem}[4]{
    \noindent
    {\PlexMedium #1} \\
    {\PlexSans #2} \\
    {\PlexLight\itshape #3} \\
    {\PlexExtraLight #4}
    \vspace{0.85cm}
}
